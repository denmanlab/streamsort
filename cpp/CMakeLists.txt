cmake_minimum_required(VERSION 3.18)
project(streamsort_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(STREAMSORT_PYTHON "Build the streamsort_core Python module" ON)
option(STREAMSORT_TESTS "Build the C++ smoke test" ON)

add_library(streamsort_core STATIC core/src/suppressor.cpp core/src/decoder.cpp core/src/bank.cpp)
target_include_directories(streamsort_core PUBLIC core/include)
set_property(TARGET streamsort_core PROPERTY POSITION_INDEPENDENT_CODE ON)
# no FMA contraction / fast-math: keeps arithmetic identical to the numba reference kernels
if(MSVC)
  target_compile_options(streamsort_core PRIVATE /O2 /fp:precise)
else()
  target_compile_options(streamsort_core PRIVATE -O3 -ffp-contract=off -fno-fast-math)
endif()

if(STREAMSORT_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(streamsort_core_py python/bindings.cpp)
  set_target_properties(streamsort_core_py PROPERTIES OUTPUT_NAME streamsort_core)
  target_link_libraries(streamsort_core_py PRIVATE streamsort_core)
  install(TARGETS streamsort_core_py DESTINATION .)
endif()

if(STREAMSORT_TESTS)
  add_executable(streamsort_core_test tests/test_core.cpp)
  target_link_libraries(streamsort_core_test PRIVATE streamsort_core)
endif()
